cmake_minimum_required(VERSION 3.20)
project(EchoHeartsServer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(echohearts-server src/server.cpp)
if(WIN32)
  target_link_libraries(echohearts-server PRIVATE ws2_32)
else()
  target_link_libraries(echohearts-server PRIVATE pthread)
endif()
